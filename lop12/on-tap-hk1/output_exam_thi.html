📊 Thiết lập lưu kết quả lên Google Sheet

Chỉ cần làm 1 lần. Sau đó mọi học sinh nộp bài sẽ tự động ghi vào 3 sheet của bạn.

📋 Sheet 1 · Tổng hợp
1 dòng / học sinh · Điểm, % đúng từng phần
🔍 Sheet 2 · Chi tiết bài làm
N dòng / HS · Từng câu: chọn gì, đáp án đúng, đúng/sai
📈 Sheet 3 · Phân tích câu hỏi
Câu nào sai nhiều nhất · Tỷ lệ đúng mỗi câu
1
Mở Google Sheet mới → menu Extensions → Apps Script
2
Xóa toàn bộ code mặc định, dán đoạn code bên dưới vào editor, rồi nhấn 💾 Save:
// ═══════════════════════════════════════════════════════════════ // Apps Script — Hệ thống thu bài thi tự động // Phiên bản: 2.0 | 3 sheet: Tổng hợp · Chi tiết · Phân tích // ═══════════════════════════════════════════════════════════════ const SS = SpreadsheetApp.getActiveSpreadsheet(); const CLR = { headerBg : '#1a237e', headerFg : '#ffffff', partMC : '#e3f2fd', partTF : '#f3e5f5', partSA : '#e8f5e9', correct : '#c8e6c9', wrong : '#ffcdd2', unanswered: '#fff9c4', rowGreen : '#e8f5e9', rowYellow : '#fff9c4', rowRed : '#ffebee', analyticHdr: '#263238', analyticFg: '#eceff1', }; // ──────────────────────────────────────────────────────────────── function doPost(e) { try { const data = JSON.parse(e.postData.contents); _writeSummaryRow(data); _writeDetailRows(data); _updateAnalytics(data); return ContentService .createTextOutput(JSON.stringify({ status: 'ok' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function doGet() { return ContentService.createTextOutput( 'OK — Apps Script đang hoạt động. Phiên bản 2.0' ); } // ════════════════════════════════════════════════════════════════ // SHEET 1: TỔNG HỢP (1 dòng / lần nộp) // Cột: STT | Thời gian | Họ tên | Lớp | Chế độ | Đề | Điểm | Tối đa // | Đúng | Tổng | % | TN đúng | TN tổng | ĐS đúng | ĐS tổng | Ngắn đúng | Ngắn tổng // ════════════════════════════════════════════════════════════════ function _writeSummaryRow(data) { let sh = SS.getSheetByName('Tổng hợp'); if (!sh) sh = SS.insertSheet('Tổng hợp', 0); if (sh.getLastRow() === 0) { const hdrs = [ 'STT','Thời gian nộp','Họ và tên','Lớp','Chế độ','Đề đã làm', 'Điểm','Tổng điểm tối đa','Đúng','Tổng câu','% Đúng', 'TN đúng','TN tổng','ĐS đúng','ĐS tổng','Ngắn đúng','Ngắn tổng' ]; sh.appendRow(hdrs); const hdr = sh.getRange(1, 1, 1, hdrs.length); hdr.setBackground(CLR.headerBg).setFontColor(CLR.headerFg) .setFontWeight('bold').setFontSize(11).setVerticalAlignment('middle'); sh.setRowHeight(1, 36); sh.setFrozenRows(1); sh.setFrozenColumns(3); const widths = [45,160,170,75,90,180, 80,100,65,80,70, 75,75,75,75,90,90]; widths.forEach((w,i) => sh.setColumnWidth(i+1, w)); sh.getRange('A:A').setHorizontalAlignment('center'); sh.getRange('G:Q').setHorizontalAlignment('center'); } // ── Tính lại điểm theo tỉ lệ: (rawScore / maxRaw) × max_score ── // Đảm bảo đúng hết 100% → luôn đạt đúng max_score const bkd = data.breakdown || {}; const scfg = data.scoring_cfg || {}; const mcPerQ_g = parseFloat(scfg.mc_per_q || 0.25); const tfScale_g = scfg.tf_scale || [0.1, 0.25, 0.5, 1.0]; const shortPerQ_g = parseFloat(scfg.short_per_q || 0.5); const maxScore = parseFloat(scfg.max_score || data.max_score || 10); // Đếm số câu thực tế từ breakdown (mcT/tfT/shortT = tổng câu từng loại) const nMC = parseInt(bkd.mcT || 0); const nTF = parseInt(bkd.tfT || 0); // số câu ĐS (mỗi câu có nhiều ý) const nShort = parseInt(bkd.shortT || 0); // maxRaw theo số câu thực tế const maxRaw_g = nMC * mcPerQ_g + nTF * tfScale_g[tfScale_g.length - 1] + nShort * shortPerQ_g; // rawScore từ bkd let rawScore_g = (parseInt(bkd.mc || 0)) * mcPerQ_g + nShort * shortPerQ_g; // sẽ cộng thêm ĐS bên dưới // Trả lời ngắn: chỉ đúng rawScore_g -= nShort * shortPerQ_g; // reset short rawScore_g += (parseInt(bkd.short || 0)) * shortPerQ_g; // ĐS: dùng data.questions nếu có, else ước lượng từ bkd.tf if (data.questions) { data.questions.filter(q => q.type === 'tf').forEach(q => { const c = parseInt(q.tfCorrect || 0); if (c > 0) rawScore_g += tfScale_g[Math.min(c, tfScale_g.length) - 1]; }); } else { // Fallback: bkd.tf = tổng số ý đúng (không biết phân bố câu) // Giả sử mỗi câu có tfScale.length ý → chia đều const rowsPerQ = tfScale_g.length; const tfIxArr = Array(nTF).fill(0); let remaining = parseInt(bkd.tf || 0); for (let i = 0; i < nTF && remaining > 0; i++) { const give = Math.min(rowsPerQ, remaining); tfIxArr[i] = give; remaining -= give; } tfIxArr.forEach(c => { if(c>0) rawScore_g += tfScale_g[Math.min(c,tfScale_g.length)-1]; }); } const computedScore = maxRaw_g > 0 ? Math.min(rawScore_g / maxRaw_g * maxScore, maxScore) : (data.score ? parseFloat(data.score) : 0); const finalScore = computedScore.toFixed(2); const correct = parseInt(data.correct) || 0; const total = parseInt(data.total) || 0; const pct = total > 0 ? (correct / total * 100).toFixed(1) + '%' : ''; const stt = sh.getLastRow(); sh.appendRow([ stt, data.timestamp, data.student, data.studentClass, data.mode, data.examLabel||'', finalScore, maxScore, correct, total, pct, hasMC ? (bkd.mc ?? '') : '—', hasMC ? (bkd.mcT ?? '') : '—', hasTF ? (bkd.tf ?? '') : '—', hasTF ? (bkd.tfT ?? '') : '—', hasSHORT ? (bkd.short ?? '') : '—', hasSHORT ? (bkd.shortT ?? '') : '—', ]); const r = sh.getLastRow(); const scoreVal = parseFloat(finalScore) || 0; const pct80 = maxScore * 0.8; const pct50 = maxScore * 0.5; const rowBg = scoreVal >= pct80 ? CLR.rowGreen : scoreVal >= pct50 ? CLR.rowYellow : CLR.rowRed; sh.getRange(r, 1, 1, 17).setBackground(rowBg); sh.getRange(r, 7) .setFontWeight('bold').setFontSize(12) .setFontColor(scoreVal >= pct80 ? '#1b5e20' : scoreVal >= pct50 ? '#e65100' : '#b71c1c'); sh.getRange(r, 11).setFontWeight('bold'); } // ════════════════════════════════════════════════════════════════ // SHEET 2: CHI TIẾT BÀI LÀM (N dòng / lần nộp, 1 dòng / câu) // Cột: STT nộp | Thời gian | Họ tên | Lớp | Điểm // | Phần | Câu | Nội dung (tóm tắt) | Học sinh chọn | Đáp án đúng | Kết quả // ════════════════════════════════════════════════════════════════ function _writeDetailRows(data) { let sh = SS.getSheetByName('Chi tiết bài làm'); if (!sh) sh = SS.insertSheet('Chi tiết bài làm'); if (sh.getLastRow() === 0) { const hdrs = [ 'STT nộp','Thời gian','Họ và tên','Lớp','Điểm', 'Phần','Câu','Nội dung câu hỏi (tóm tắt)', 'Học sinh chọn','Đáp án đúng','Kết quả' ]; sh.appendRow(hdrs); const hdr = sh.getRange(1, 1, 1, hdrs.length); hdr.setBackground(CLR.headerBg).setFontColor(CLR.headerFg) .setFontWeight('bold').setFontSize(11).setVerticalAlignment('middle'); sh.setRowHeight(1, 36); sh.setFrozenRows(1); sh.setFrozenColumns(4); const widths = [65,155,165,70,65, 165,95,300,110,110,90]; widths.forEach((w,i) => sh.setColumnWidth(i+1, w)); sh.getRange('I:K').setHorizontalAlignment('center'); } const qs = data.questions || []; if (!qs.length) return; // STT nộp = số dòng dữ liệu đã có ở Sheet Tổng hợp const sumSh = SS.getSheetByName('Tổng hợp'); const stt = sumSh ? Math.max(1, sumSh.getLastRow() - 1) : '?'; const rows = qs.map(q => [ stt, data.timestamp, data.student, data.studentClass, data.score, q.part || '', q.qnum || '', q.qtext || '', q.userAns || '—', q.correctAns || '', q.result || '' ]); if (!rows.length) return; const startRow = sh.getLastRow() + 1; sh.getRange(startRow, 1, rows.length, rows[0].length).setValues(rows); // Tô màu từng dòng rows.forEach((row, i) => { const r = startRow + i; const res = row[10]; // cột K — Kết quả const ptype = (qs[i].type || '').toLowerCase(); // Cột Phần: màu theo loại const partBg = ptype === 'mc' ? CLR.partMC : ptype === 'tf' ? CLR.partTF : CLR.partSA; sh.getRange(r, 6).setBackground(partBg).setFontWeight('bold'); // Cột Kết quả: màu + chữ đậm const resBg = res === 'Đúng' ? CLR.correct : res === 'Sai' ? CLR.wrong : CLR.unanswered; const resFg = res === 'Đúng' ? '#1b5e20' : res === 'Sai' ? '#b71c1c' : '#795548'; sh.getRange(r, 11).setBackground(resBg).setFontWeight('bold').setFontColor(resFg); // Học sinh sai / chưa làm: làm nổi đáp án đúng if (res === 'Sai' || res === 'Chưa làm') { sh.getRange(r, 10).setFontWeight('bold').setFontColor('#1b5e20'); sh.getRange(r, 9).setFontColor('#b71c1c'); } }); // Viền nhóm học sinh (khung ngoài đậm hơn) const range = sh.getRange(startRow, 1, rows.length, rows[0].length); range.setBorder(false, false, false, false, true, false, '#bdbdbd', SpreadsheetApp.BorderStyle.SOLID); range.setBorder(true, true, true, true, null, null, '#3949ab', SpreadsheetApp.BorderStyle.SOLID_MEDIUM); } // ════════════════════════════════════════════════════════════════ // SHEET 3: PHÂN TÍCH CÂU HỎI // Tự động cập nhật: câu nào sai nhiều nhất, tỷ lệ đúng mỗi câu // Cột: Phần | Câu | Tổng HS làm | Đúng | Sai | Chưa làm | % Đúng | Đánh giá // ════════════════════════════════════════════════════════════════ function _updateAnalytics(data) { let sh = SS.getSheetByName('Phân tích câu hỏi'); if (!sh) sh = SS.insertSheet('Phân tích câu hỏi'); if (sh.getLastRow() === 0) { const hdrs = ['Phần','Câu hỏi','Tổng HS làm','Đúng','Sai','Chưa làm','% Đúng','Đánh giá']; sh.appendRow(hdrs); const hdr = sh.getRange(1, 1, 1, hdrs.length); hdr.setBackground(CLR.analyticHdr).setFontColor(CLR.analyticFg) .setFontWeight('bold').setFontSize(11).setVerticalAlignment('middle'); sh.setRowHeight(1, 36); sh.setFrozenRows(1); const widths = [165,110,100,70,70,90,80,120]; widths.forEach((w,i) => sh.setColumnWidth(i+1, w)); sh.getRange('C:H').setHorizontalAlignment('center'); } const qs = data.questions || []; if (!qs.length) return; // Đọc toàn bộ dữ liệu hiện có để tìm / cập nhật dòng câu hỏi const lastRow = sh.getLastRow(); const allData = lastRow > 1 ? sh.getRange(2, 1, lastRow - 1, 8).getValues() : []; // Map: "Phần|Câu" → rowIndex (0-based trong allData) const keyMap = {}; allData.forEach((row, i) => { keyMap[row[0] + '|' + row[1]] = i; }); qs.forEach(q => { const key = (q.part || '') + '|' + (q.qnum || ''); const res = (q.result || '').trim(); const idx = keyMap[key]; if (idx !== undefined) { // Cập nhật dòng đã có const row = allData[idx]; const total = (parseInt(row[2]) || 0) + 1; const correct = (parseInt(row[3]) || 0) + (res === 'Đúng' ? 1 : 0); const wrong = (parseInt(row[4]) || 0) + (res === 'Sai' ? 1 : 0); const unanswer = (parseInt(row[5]) || 0) + (res === 'Chưa làm' ? 1 : 0); const pct = total > 0 ? Math.round(correct / total * 100) + '%' : ''; const eval_ = total === 0 ? '' : correct/total >= 0.8 ? '✅ Dễ' : correct/total >= 0.5 ? '⚠️ Trung bình' : '❌ Khó'; const shRow = idx + 2; // +1 header +1 0-based sh.getRange(shRow, 3, 1, 6).setValues([[total, correct, wrong, unanswer, pct, eval_]]); // Tô màu ô % Đúng const pctVal = correct / total; sh.getRange(shRow, 7) .setBackground(pctVal >= 0.8 ? '#c8e6c9' : pctVal >= 0.5 ? '#fff9c4' : '#ffcdd2') .setFontWeight('bold') .setFontColor(pctVal >= 0.8 ? '#1b5e20' : pctVal >= 0.5 ? '#e65100' : '#b71c1c'); } else { // Thêm dòng mới const isCorrect = res === 'Đúng' ? 1 : 0; const isWrong = res === 'Sai' ? 1 : 0; const isUnanswer = res === 'Chưa làm' ? 1 : 0; const eval_ = res === 'Đúng' ? '✅ Dễ' : res === 'Sai' ? '❌ Khó' : '⚠️ Trung bình'; const newRow = [ q.part || '', q.qnum || '', 1, isCorrect, isWrong, isUnanswer, isCorrect === 1 ? '100%' : '0%', eval_ ]; sh.appendRow(newRow); const r = sh.getLastRow(); // Màu cột Phần const ptype = (q.type || '').toLowerCase(); const partBg = ptype === 'mc' ? CLR.partMC : ptype === 'tf' ? CLR.partTF : CLR.partSA; sh.getRange(r, 1).setBackground(partBg).setFontWeight('bold'); // Màu % Đúng sh.getRange(r, 7) .setBackground(isCorrect ? '#c8e6c9' : '#ffcdd2') .setFontWeight('bold') .setFontColor(isCorrect ? '#1b5e20' : '#b71c1c'); // Cập nhật keyMap keyMap[key] = allData.length; allData.push(newRow); } }); }
